# ITX boot menu: black screen, two entries, no decoration.
#
# The two labels are images (see make-images.sh), because GRUB cannot load
# custom fonts when Secure Boot is on. GRUB always draws at least three rows,
# so 06_itx adds an empty third entry. Ubuntu's standard entries (older
# kernels, recovery, firmware settings) follow it and appear when you press
# Down past the empty row.

title-text: ""
desktop-color: "#000000"
terminal-font: "Unifont Regular 16"

    left = 50%-92
    top = 50%-56
    width = 760
    height = 160
    item_font = "Unifont Regular 16"
    item_color = "#6a6a6a"
    selected_item_color = "#f0f0f0"
    icon_width = 184
    icon_height = 48
    item_height = 48
    item_padding = 0
    item_icon_space = 16
    item_spacing = 8
    selected_item_pixmap_style = "select_*.png"
    scrollbar = false
}

+ label {
    id = "__timeout__"
    text = "%d"
    left = 50%-72
    top = 50%+72
    width = 160
    align = "left"
    color = "#3a3a3a"
    font = "Unifont Regular 16"
}
